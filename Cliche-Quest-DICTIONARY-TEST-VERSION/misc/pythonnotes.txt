# DICTIONARIES
# dictionaries in real life map (or associate) words with definitions
# what is a definition?
# a statement describing what a word means
# in other words, a definition maps a word to a sentence

# dictionaries in coding are really similar
# dictionaries map certain data to other data

# example:
dictionary = {}
dictionary['mean'] = "Having the quality of being disrespectful."
dictionary['apple'] = "A sweet fruit which is typically red and round."

# dictionaries are especially useful when you want to make "data tables"
scores = {}
scores['Dave'] = 100
scores['Nate'] = 90
scores['Link'] = 50

# in dictionaries, the original data is called a "key"
# and the data it is attached to is called the "value"

# SYNTAX
# to add or change an entry in a dictionary:
# dictionary[key] = value

# to get a value from the dictionary given a key:
# dictionary[key]
# will replace that with its value once it runs, i.e.

print(scores['Dave']) # prints 100

# dictionaries can map ANY key to ANY value
employees = {}
employees['Bob'] = ['2020-11-20', 18.60, 30]
employees['Daisy'] = ['2022-01-30', 20.20, 28]


# dictionaries are ONE WAY
# meaning keys are mapped to values, but not vice versa
# so i can ask a dictionary to get the value associated with a key
# but i cannot "ask" a dictionary to get the key associated with a value

# the reason for this is VERY simple
# it is possible that two keys get mapped to the same value
# using the school grades above, we could have two students with a score of 100
scores['Billy'] = 100

# now there are two keys in the dictionary that are mapped to 100
# so its simply impossible to inquire like "what is the key associated with 100"
